    fun erzeugen() {
        val ziel = File(System.getProperty("ziel")).apply { mkdirs() }
        val kurs = schwimmkurs()

        val kursText = KCodec.encode(kurs)
        val gruppenText = GCodec.encode(gruppen)
        File(ziel, "test-schwimmkurs.json").writeText(kursText)
        File(ziel, "test-schwimmgruppen.json").writeText(gruppenText)

        // Gegenprobe: lassen sich die Dateien wieder lesen?
        KCodec.decode(kursText); GCodec.decode(gruppenText)
        val leitung = schwimmleitung(kursText, gruppenText)

    // ---------------- Schwimmkurs ----------------

    private fun schwimmkurs(): KBestand {
        val kinder = mutableListOf<KKind>()
        for (i in 1..72) {
            val w = r.nextBoolean()
            val nach = nachnamen.zufall()
            val (ort, plz) = when (r.nextInt(10)) { in 0..5 -> orte.take(9).zufall(); 6, 7 -> orte.subList(9, 12).zufall(); 8 -> orte.takeLast(2).zufall(); else -> "" to "" }
            val geburt = tag(LocalDate.of(2018, 9, 1), LocalDate.of(2022, 3, 31))
            kinder += KKind(
                id = "kind-$i", vorname = vorname(w), nachname = nach, geburtsdatum = geburt,
                eltern = erwachsene.zufall() + " " + nach, email = "", telefon = tel(),
                strasse = if (ort.isBlank()) "" else "${strassen.zufall()} ${1 + r.nextInt(40)}", plz = plz, ort = ort,
                anmeldedatum = tag(LocalDate.of(2024, 9, 1), LocalDate.of(2026, 10, 15)),
                notiz = if (ort.isBlank() && r.nextBoolean()) "Kindergarten ${listOf("Fischach", "Willmatshofen", "Langenneufnach").zufall()}" else "",
                einschulung = (geburt.year + 6 + if (geburt.monthValue >= 7) 1 else 0).toString(),
                listenplatz = i,
                pausiert = r.nextInt(25) == 0,
            ).let { it.copy(email = mail(it.eltern.substringBefore(' '), nach)) }
        }
        kinder.sortBy { it.anmeldedatum }
        val samstag11 = LocalTime.of(11, 0) to LocalTime.of(11, 45)
        val starts = listOf(
            LocalDate.of(2025, 9, 20), LocalDate.of(2025, 11, 22), LocalDate.of(2026, 1, 24), LocalDate.of(2026, 3, 21), LocalDate.of(2026, 5, 9),
            LocalDate.of(2026, 9, 19), LocalDate.of(2026, 11, 21), LocalDate.of(2027, 1, 23),
        )
        val frei = kinder.filter { !it.pausiert }.toMutableList()
        val kurse = starts.mapIndexed { i, start ->
            val vergangen = start.isBefore(LocalDate.of(2026, 9, 1))
            val saison = if (start.monthValue >= 9) start.year else start.year - 1
            val nr = if (vergangen) i + 1 else i - 4
            val kuerzel = "SK%02d %02d%02d".format(nr, saison % 100, (saison + 1) % 100)
            val anzahl = when { start.isAfter(LocalDate.of(2027, 1, 1)) -> 4; else -> 7 + r.nextInt(2) }
            val tn = frei.filter { (it.anmeldedatum ?: heute).isBefore(start.minusDays(14)) && it.geburtsdatum!!.plusYears(5).isBefore(start) }
                .sortedBy { it.geburtsdatum }.take(anzahl)
            frei.removeAll(tn.toSet())
            KKurs(
                id = "kurs-${i + 1}", name = "Seepferdchenkurs $kuerzel", kuerzel = kuerzel, plaetze = 8, ziel = "Seepferdchen", minAlter = 5,
                ort = "Hallenbad Fischach", kursleitung = listOf("Sabine Kugler", "Thomas Reiter").zufall(),
                termine = Terminserie.woechentlich(start, DayOfWeek.SATURDAY, 8, samstag11.first, samstag11.second),
                teilnehmer = tn.map { it.id },
            )
        }
        // Abgeschlossene Kurse: die meisten haben das Seepferdchen geschafft.
        val geschafft = kurse.filter { it.termine.last().datum.isBefore(heute) }.flatMap { it.teilnehmer }.filter { r.nextInt(10) < 8 }.toSet()
        val fertig = kinder.map { if (it.id in geschafft) it.copy(abzeichen = setOf("Seepferdchen")) else it }
        return KBestand(
            kinder = fertig, kurse = kurse,
            einstellungen = KEinstellungen(iban = "DE00 1234 5678 9012 3456 78", kontoinhaber = "Testverein Schwimmen", absender = "Ihr Test-Team Schwimmen"),
        )
    }

    // ---------------- Schwimmgruppen ----------------

    private fun schwimmgruppen(): GBestand {
        var b = GBestand()
        val altersgruppe = mapOf(
            "gruppe-1" to (6..7), "gruppe-1-6" to (6..7), "gruppe-2" to (8..9), "gruppe-2-3" to (8..9), "gruppe-2-6" to (8..9),
            "gruppe-3" to (10..11), "gruppe-4" to (12..15), "kader" to (12..17),
        )
        val mitglieder = mutableListOf<GMitglied>()
        var nr = 1000
        for ((gid, alter) in altersgruppe) {
            val anzahl = if (gid == "kader") 6 else 6 + r.nextInt(5)
            repeat(anzahl) {
                val w = r.nextBoolean()
                val nach = nachnamen.zufall()
                val a = alter.random(r)
                val geburt = tag(heute.minusYears(a + 1L).plusDays(1), heute.minusYears(a.toLong()))
                val (ort, plz) = orte.take(10).zufall()
                val vor = vorname(w)
                val gruppenSet = if (gid == "kader") setOf("kader", listOf("gruppe-3", "gruppe-4").zufall()) else setOf(gid)
                mitglieder += GMitglied(
                    id = "m-${nr}", vorname = vor, nachname = nach, geburtsdatum = geburt, geschlecht = if (w) "w" else "m",
                    mitgliedsnummer = (nr++).toString(), email = mail(erwachsene.zufall(), nach), telefon = tel(),
                    strasse = "${strassen.zufall()} ${1 + r.nextInt(40)}", plz = plz, ort = ort,
                    eintritt = (geburt.year + 6 + r.nextInt(3)).coerceAtMost(2026).toString(),
                    fotorechte = r.nextInt(8) != 0, gesundheitserklaerung = r.nextInt(10) != 0,
                    gruppen = gruppenSet,
                    interesse = when { gid == "kader" -> GInteresse.JA; a >= 9 -> GInteresse.entries.zufallE(); else -> GInteresse.OFFEN },
                )
            }
        }
        // Ein paar noch ohne Gruppe und zwei Ausgetretene
        repeat(4) { val w = r.nextBoolean(); val nach = nachnamen.zufall(); mitglieder += GMitglied("m-${nr}", vorname(w), nach, tag(LocalDate.of(2017, 1, 1), LocalDate.of(2019, 12, 31)), if (w) "w" else "m", (nr++).toString(), eintritt = "2026") }
        repeat(2) { i -> mitglieder[i * 7] = mitglieder[i * 7].copy(aktiv = false, gruppen = emptySet()) }

        // Erfolge und Zeiten
        val abz = listOf("Seepferdchen", "Bronze", "Silber", "Gold")
        val mitErfolgen = mitglieder.map { m ->
            val a = m.alterAm(heute) ?: 8
            val stufe = when { a <= 7 -> 1; a <= 9 -> 2; a <= 11 -> 3; else -> 4 }.coerceAtMost(abz.size) - if (r.nextInt(4) == 0) 1 else 0
            val erfolge = abz.take(stufe.coerceAtLeast(1)).mapIndexed { i, t -> GErfolg("e-${m.id}-$i", t, GErfolgArt.ABZEICHEN, null) }.toMutableList()
            val zeiten = mutableListOf<GZeit>()
            if ("kader" in m.gruppen || "gruppe-4" in m.gruppen) {
                for (strecke in listOf("50 m Freistil", "50 m Brust", "100 m Freistil").shuffled(r).take(2)) {
                    var basis = (when (strecke) { "100 m Freistil" -> 9000; "50 m Brust" -> 5200; else -> 4200 } - a * 150 + r.nextInt(600))
                    var d = LocalDate.of(2025, 10, 1)
                    repeat(3 + r.nextInt(3)) { i ->
                        zeiten += GZeit("z-${m.id}-$strecke-$i", strecke, basis, d, wettkampf = i % 2 == 1, anlass = if (i % 2 == 1) listOf("Bezirksmeisterschaft", "Vereinsvergleich Bobingen", "Kreismeisterschaft").zufall() else "")
                        basis -= r.nextInt(-40, 160); d = d.plusDays(40L + r.nextInt(60))
                    }
                }
                if (r.nextBoolean()) erfolge += GErfolg("w-${m.id}", "${1 + r.nextInt(3)}. Platz 50 m ${listOf("Freistil", "Brust", "Rücken").zufall()}, Bezirksmeisterschaft", GErfolgArt.WETTKAMPF, LocalDate.of(2026, 3, 14))
            }
            m.copy(erfolge = erfolge, zeiten = zeiten)
        }
        mitErfolgen.forEach { b = b.mitgliedSetzen(it) }

        // Trainer: drei Erwachsene ohne Mitgliedschaft, zwei Jungtrainer aus dem Kader, ein Helfer
        val kader = mitErfolgen.filter { "kader" in it.gruppen && it.aktiv }.sortedBy { it.geburtsdatum }
        val trainer = listOf(
            GTrainer("t-1", null, "Sabine Kugler", LocalDate.of(1979, 4, 12), GRang.TRAINER, "Trainer C Schwimmen", tel()),
            GTrainer("t-2", null, "Thomas Reiter", LocalDate.of(1985, 8, 3), GRang.TRAINER, "Trainer C Schwimmen", tel()),
            GTrainer("t-3", null, "Katrin Lechner", LocalDate.of(1992, 1, 27), GRang.TRAINER, "Rettungsschwimmer Silber", tel()),
            GTrainer("t-4", kader[0].id, rang = GRang.JUNGTRAINER, schein = "Rettungsschwimmer Bronze"),
            GTrainer("t-5", kader[1].id, rang = GRang.JUNGTRAINER),
            GTrainer("t-6", null, "Markus Vogl", LocalDate.of(1970, 11, 2), GRang.HELFER, "", tel()),
        )
        trainer.forEach { b = b.trainerSetzen(it) }
        val stamm = mapOf(
            "gruppe-1" to listOf("t-3", "t-5"), "gruppe-1-6" to listOf("t-1", "t-4"), "gruppe-2" to listOf("t-2"), "gruppe-2-3" to listOf("t-3"),
            "gruppe-2-6" to listOf("t-1", "t-5"), "gruppe-3" to listOf("t-2", "t-4"), "gruppe-4" to listOf("t-2"), "kader" to listOf("t-1", "t-2"),
            "wassergymnastik" to listOf("t-6"), "erwachsene" to listOf("t-3"),
        )
        b = b.copy(gruppen = b.gruppen.map { g -> g.copy(trainerIds = stamm[g.id].orEmpty(), plaetze = if (g.art.name == "EXTERN") 15 else 12) })

        // Trainings von Saisonbeginn 2025 bis gestern, ohne Ferien
        val ferien = listOf(
            LocalDate.of(2025, 10, 27) to LocalDate.of(2025, 11, 7), LocalDate.of(2025, 12, 22) to LocalDate.of(2026, 1, 5),
            LocalDate.of(2026, 2, 16) to LocalDate.of(2026, 2, 20), LocalDate.of(2026, 3, 30) to LocalDate.of(2026, 4, 10),
            LocalDate.of(2026, 5, 26) to LocalDate.of(2026, 6, 5), LocalDate.of(2026, 8, 1) to LocalDate.of(2026, 9, 14),
        )
        var d = LocalDate.of(2025, 9, 15)
        val einheiten = mutableListOf<GEinheit>()
        while (d.isBefore(heute)) {
            if (ferien.none { (von, bis) -> !d.isBefore(von) && !d.isAfter(bis) }) {
                for (g in b.gruppen) for (z in g.zeiten) {
                    if (z.wochentag != d.dayOfWeek.value) continue
                    if (r.nextInt(12) == 0) continue // nicht erfasst
                    val ausgefallen = r.nextInt(30) == 0
                    val leute = b.mitgliederIn(g.id).filter { r.nextInt(10) < 7 }.map { it.id }.toSet()
                    val vertretung = if (!ausgefallen && r.nextInt(10) == 0) g.trainerIds.firstOrNull()?.let { mapOf(it to listOf("t-1", "t-2", "t-3", "t-6").filter { t -> t != it }.zufall()) } else null
                    einheiten += GEinheit(
                        gruppeId = g.id, datum = d, beginn = z.beginn, ende = z.ende,
                        anwesend = if (ausgefallen || g.nurZaehlen) emptySet() else leute,
                        weitere = when { ausgefallen -> 0; g.nurZaehlen -> 6 + r.nextInt(9); r.nextInt(8) == 0 -> 1; else -> 0 },
                        ausgefallen = ausgefallen,
                        vermerk = when { ausgefallen -> listOf("Bad gesperrt (Technik)", "Trainer krank", "Hallenreinigung").zufall(); r.nextInt(25) == 0 -> listOf("Bahn 3 gesperrt", "Abzeichenprüfung", "Probeschwimmer da", "Staffeltraining").zufall(); else -> "" },
                        vertretungen = vertretung.orEmpty(),
                    )
                }
            }
            d = d.plusDays(1)
        }
        return b.copy(einheiten = einheiten)
    }

        d = d.setzen(k, jetzt).setzen(g, jetzt)
        var n = 0
        d = d.trainerUebernehmen(g, jetzt) { "p-${++n}" }.first
        d = d.setzen(Person("p-leitung", "Florian Wiedemann", "Abteilungsleitung", tel()), jetzt)

            .setzen(Person("p-kasse", "Daniel Probst", "Kassenwart", tel()), jetzt)

        // Nachweise

            Material("mat-stange", "Rettungsstange", "Wand Beckenrand", 1, pruefen = LocalDate.of(2026, 11, 15)),
            Material("mat-eh", "Erste-Hilfe-Kasten", "Schrank Trainer", 1, pruefen = LocalDate.of(2026, 10, 31), notiz = "Pflaster fast leer"),
